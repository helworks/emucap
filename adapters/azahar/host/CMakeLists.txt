cmake_minimum_required(VERSION 3.16)
project(emucap_azahar_host CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
add_executable(emucap-azahar-host emucap_libretro_host.cpp "${EMUCAP_COMMON}/emucap_producer.cpp")
target_include_directories(emucap-azahar-host PRIVATE "${EMUCAP_COMMON}" "${LIBRETRO_INCLUDE}")
target_link_libraries(emucap-azahar-host PRIVATE ZLIB::ZLIB Threads::Threads ${CMAKE_DL_LIBS})
if (EMUCAP_BUILD_ID)
    target_compile_definitions(emucap-azahar-host PRIVATE "EMUCAP_BUILD_ID=\"${EMUCAP_BUILD_ID}\"")
endif()
set_target_properties(emucap-azahar-host PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
